def test_compra_en_efectivo(repo_falso):
    # Arrange
    pedido = pedido_valido(cantidad=3, edades=[40, 35, 10], tipo_pase="VIP")
    # Act
    compra = comprar(pedido, repo_falso)
    # Assert: dos entradas completas y una con descuento del 50%.
    assert compra.id == 1
    assert compra.cantidad == 3
    assert compra.fecha == fecha(2026, 10, 10)
    assert compra.total == 50000
    assert compra.estado == "PENDIENTE_PAGO_BOLETERIA"
    assert longitud(repo_falso.compras) == 1
    assert longitud(repo_falso.mails) == 1
    destinatario, mensaje = repo_falso.mails[0]
    assert destinatario == pedido.email
    assert mensaje contiene "3 entradas", "10/10/2026", "$50000", "boletería"

PEDIDO = diccionario de pedido_valido()

def test_api_registra_compra_y_mail(cliente, repo_sqlite):
    respuesta = cliente.POST("/api/compras", json=PEDIDO)
    assert respuesta.status == 201
    assert respuesta.json.total == 15000
    assert respuesta.json.cantidad == 2
    assert respuesta.json.fecha == FECHA_ABIERTO
    assert respuesta.json.estado == "PENDIENTE_PAGO_BOLETERIA"
    assert longitud(repo_sqlite.listar_compras()) == 1
    assert longitud(repo_sqlite.listar_mails()) == 1

def test_api_rechaza_sin_guardar(cliente, repo_sqlite):
    respuesta = cliente.POST("/api/compras",
        json=PEDIDO con fecha=FECHA_CERRADO y cantidad=-1)
    assert respuesta.status == 422
    assert respuesta.json.errores contiene mensajes de parque cerrado y cantidad
    assert repo_sqlite.listar_compras() == []
    assert repo_sqlite.listar_mails() == []

def test_limite_acumulado(cliente, repo_sqlite):
    pedido = PEDIDO con cantidad=5 y edades=[3]*5
    assert cliente.POST("/api/compras", json=pedido).status == 201
    assert cliente.POST("/api/compras", json=pedido).status == 201
    # Se alcanzaron diez entradas, aunque sean gratuitas.
    respuesta = cliente.POST("/api/compras", json=PEDIDO con email="otro@mail.com")
    assert respuesta.status == 422
    assert longitud(repo_sqlite.listar_compras()) == 2
    assert longitud(repo_sqlite.listar_mails()) == 2
    # El límite es por fecha, no por toda la historia del usuario.
    assert cliente.POST("/api/compras", json=PEDIDO con fecha="2026-10-11").status == 201

def test_identidad_de_sesion(cliente):
    sesion = cliente.GET("/api/sesion").json
    respuesta = cliente.POST("/api/compras", json=PEDIDO con email="otro@mail.com")
    assert respuesta.json.email == sesion.email

El servidor fija el titular de la sesión mock. No se prueba un login real.
El correo enviado en el pedido no reemplaza la identidad de esa sesión.
El límite acumulado se comprueba en una transacción SQLite; este test secuencial
no constituye una prueba específica de concurrencia.
